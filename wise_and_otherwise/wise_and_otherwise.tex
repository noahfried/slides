          \node[star, star points=5, star point ratio=2.618, fill=white, inner sep=0pt,

    }

\begin{frame}{What is a proverb?}
  \begin{itemize}\setlength{\itemsep}{0.7em}
    \item A short, traditional saying.
    \item It is used to represent a certain situation.
    \item It means more than just its words.
  \end{itemize}
  \vspace{1.2em}
  Eg: \alert{The apple doesn't fall far from the tree.}\\[1.2em]
  Can you think of any examples in German? --- or in English?
\end{frame}

\begin{frame}{Examples of proverbs}
  \large
  ``Actions speak louder than words.''\\[1em]
  ``A journey of a thousand miles begins with a single step.''\\[1em]
  ``Don't judge a book by its cover.''\\[1em]
  ``While the grass grows, the horse starves.''
\end{frame}

\begin{frame}{The game}
  \begin{itemize}\setlength{\itemsep}{0.55em}
    \item I'll display the beginning of a proverb (an old saying).
    \item In your group, brainstorm a believable ending to the proverb.
    \item When you're done, write it down on paper and bring it up to me.
    \item I'll read all the endings aloud --- including the real one.
    \item In your group, discuss what you think the real ending is.
    \item If you guess the correct proverb, you get one point. If you trick a group --- they vote for your proverb --- you get a point.
    \item The group with the most points at the end wins.
  \end{itemize}
\end{frame}

\begin{frame}{Example}
  There is an old English proverb:\\[0.7em]
  \textbf{``Cheese and money should always...''}\\[0.4em]
  \begin{itemize}\setlength{\itemsep}{0.55em}
    \item[] Option 1: ... \emph{attract attention.}
    \item[] Option 2: ... \emph{grow over time.}
    \item[] Option 3: ... \emph{be kept away from children.}
    \item[] Option 4: ... \emph{get counted twice.}
    \item[] Option 5: ... \emph{reflect your character.}
  \end{itemize}
\end{frame}

\begin{frame}{Example: answer}
  There is an old English proverb:\\[0.7em]
  \textbf{``Cheese and money should always be kept away from children.''}\\[0.4em]
  \begin{itemize}\setlength{\itemsep}{0.55em}
    \item[] \leavevmode\color{muted}\sout{Option 1: ... \textit{attract attention.}}
    \item[] \leavevmode\color{muted}\sout{Option 2: ... \textit{grow over time.}}
    \item[] \leavevmode\color{oldred}\textbf{Option 3: ... \textit{be kept away from children.}}
    \item[] \leavevmode\color{muted}\sout{Option 4: ... \textit{get counted twice.}}
    \item[] \leavevmode\color{muted}\sout{Option 5: ... \textit{reflect your character.}}
  \end{itemize}
\end{frame}
